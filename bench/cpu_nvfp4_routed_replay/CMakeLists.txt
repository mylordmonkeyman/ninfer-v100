cmake_minimum_required(VERSION 3.20)
project(ninfer_cpu_nvfp4_routed_replay LANGUAGES CXX)
find_package(Threads REQUIRED)
set(impl ${CMAKE_CURRENT_LIST_DIR}/../../src/targets/qwen3_8_flash_next/impl)
add_library(cpu_experts STATIC ${impl}/cpu_expert_reference.cpp
  ${impl}/cpu_expert_avx2.cpp ${impl}/cpu_expert_pool.cpp)
target_include_directories(cpu_experts PUBLIC ${CMAKE_CURRENT_LIST_DIR}/../../src)
target_compile_features(cpu_experts PUBLIC cxx_std_20)
target_link_libraries(cpu_experts PUBLIC Threads::Threads)
add_executable(ninfer_cpu_nvfp4_routed_replay main.cpp)
target_link_libraries(ninfer_cpu_nvfp4_routed_replay PRIVATE cpu_experts)
add_executable(ninfer_cpu_expert_pool_test pool_test.cpp)
target_link_libraries(ninfer_cpu_expert_pool_test PRIVATE cpu_experts)
enable_testing()
add_test(NAME cpu_expert_pool COMMAND ninfer_cpu_expert_pool_test)
